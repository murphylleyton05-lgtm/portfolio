SELECT c.nombre_cliente, v.fecha_venta, v.total_venta FROM ventas AS v JOIN clientes AS c ON v.id_cliente = c.id_cliente WHERE v.fecha_venta > (SELECT MAX(fecha_venta) FROM ventas) - INTERVAL '30 days' ORDER BY v.fecha_venta DESC;
